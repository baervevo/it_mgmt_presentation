\section{Wprowadzenie}
% ============================================================

\begin{frame}{Co to jest zarz\k{a}dzanie konfiguracj\k{a}?}
    \begin{block}{Definicja}
        Zarz\k{a}dzanie konfiguracj\k{a} (CM) to proces \'{s}ledzenia i kontrolowania zmian
        w oprogramowaniu, infrastrukturze i \'{s}rodowiskach systemowych.
    \end{block}
    \vspace{0.5em}
    \begin{itemize}
        \item Wywodzi si\k{e} z praktyk ITIL i wojskowych standard\'ow zarz\k{a}dzania
        \item W erze DevOps sta\l{}o si\k{e} fundamentem automatyzacji IT
        \item Umo\.zliwia \textbf{Infrastructure as Code} (IaC)
        \item Kluczowe poj\k{e}cia: \textit{idempotentno\'s\'c}, \textit{desired state}, \textit{drift detection}
    \end{itemize}
    \vspace{0.5em}
\end{frame}

\begin{frame}{Cele i funkcje system\'ow CM}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Cele strategiczne}
            \begin{itemize}
                \item Sp\'ojno\'s\'c \'{s}rodowisk (dev/test/prod)
                \item Redukcja b\l{}\k{e}d\'ow ludzkich
                \item Audytowalno\'s\'c zmian
                \item Szybki rollback
                \item Dokumentacja infrastruktury w kodzie
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Funkcje operacyjne}
            \begin{itemize}
                \item Automatyzacja wdro\.ze\'n
                \item Egzekwowanie stanu konfiguracji
                \item Wykrywanie dryftu konfiguracji
                \item Zarz\k{a}dzanie sekretami
                \item Raportowanie zgodno\'sci
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.8em}
\end{frame}

\begin{frame}{Struktura oprogramowania CM}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Model Push}
            \begin{itemize}
                \item Kontroler wysy\l{}a konfiguracj\k{e}
                \item Natychmiastowe dzia\l{}anie
                \item Przyk\l{}ady: Ansible, Terraform
            \end{itemize}
            \vspace{0.5em}
            \textbf{Agentless}
            \begin{itemize}
                \item Komunikacja przez SSH/WinRM
                \item Brak dodatkowego oprogramowania
                \item \L{}atwiejsze wdro\.zenie
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Model Pull}
            \begin{itemize}
                \item Agent sam pobiera konfiguracj\k{e}
                \item Ci\k{a}g\l{}e egzekwowanie stanu
                \item Przyk\l{}ady: Puppet, Chef
            \end{itemize}
            \vspace{0.5em}
            \textbf{Agent-based}
            \begin{itemize}
                \item Demon dzia\l{}a na w\k{e}\'zle
                \item Lepsza skalowalno\'s\'c
                \item Sta\l{}y monitoring stanu
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\section{Przegl\k{a}d narz\k{e}dzi}
% ============================================================

\begin{frame}{Git}
    \begin{columns}[T]
        \begin{column}{0.63\textwidth}
            \begin{itemize}
                \item \textbf{Rok}: 2005, Linus Torvalds
                \item \textbf{Typ}: Rozproszony system kontroli wersji
                \item \textbf{Licencja}: GPL v2
                \item \textbf{Hostingi}: GitHub, GitLab, Bitbucket
            \end{itemize}
            \vspace{0.4em}
            \textbf{Kluczowe funkcje:}
            \begin{itemize}
                \item \'{S}ledzenie historii zmian w plikach (diff, blame)
                \item Rozga\l{}\k{e}zianie (\textit{branching}) i \l{}\k{a}czenie (\textit{merging})
                \item Praca offline -- pe\l{}ne repozytorium lokalnie
                \item Podstawa dla CI/CD i GitOps
                \item Tagi, hooki, submodule
            \end{itemize}
        \end{column}
        \begin{column}{0.34\textwidth}
            \begin{block}{Zastosowanie}
                \begin{itemize}
                    \item Kod \'zr\'od\l{}owy
                    \item Pliki konfiguracyjne
                    \item Dokumentacja
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Ansible}
    \begin{columns}[T]
        \begin{column}{0.63\textwidth}
            \begin{itemize}
                \item \textbf{Rok}: 2012; Red Hat przej\k{a}\l{} w 2015
                \item \textbf{Typ}: Agentless CM, model push
                \item \textbf{Licencja}: GPL v3
                \item \textbf{J\k{e}zyk}: YAML
            \end{itemize}
            \vspace{0.4em}
            \textbf{Kluczowe funkcje:}
            \begin{itemize}
                \item Komunikacja przez SSH -- brak agent\'ow
                \item Idempotentne modu\l{}y
                \item Role i kolekcje zapewniające modularno\'s\'c
                \item Ansible Tower/AWX -- interfejs webowy
            \end{itemize}
        \end{column}
        \begin{column}{0.34\textwidth}
            \begin{block}{Zastosowanie}
                \begin{itemize}
                    \item Konfiguracja serwer\'ow
                    \item Deployments
                    \item Orkiestracja
                    \item Patch management
                    \item Provisioning VM
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Puppet}
    \begin{columns}[T]
        \begin{column}{0.63\textwidth}
            \begin{itemize}
                \item \textbf{Rok}: 2005, Puppet Labs (obecnie Perforce)
                \item \textbf{Typ}: Agent-based CM, model pull
                \item \textbf{Licencja}: Apache
                \item \textbf{J\k{e}zyk}: Puppet DSL
            \end{itemize}
            \vspace{0.4em}
            \textbf{Kluczowe funkcje:}
            \begin{itemize}
                \item Deklaratywny -- opisujesz \textit{stan}, nie kroki
                \item Puppet Master + Agent (katalog zasob\'ow)
                \item PuppetDB -- historia zmian i fakt\'ow
                \item Silne raportowanie i audyt
                \item Hiera -- hierarchia danych konfiguracyjnych
            \end{itemize}
        \end{column}
        \begin{column}{0.34\textwidth}
            \begin{block}{Zastosowanie}
                \begin{itemize}
                    \item Du\.ze enterprise
                    \item Sta\l{}a zgodno\'s\'c
                    \item Compliance
                    \item Setki w\k{e}z\l{}\'ow
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Chef}
    \begin{columns}[T]
        \begin{column}{0.63\textwidth}
            \begin{itemize}
                \item \textbf{Rok}: 2009, Opscode; Progress Software od 2020
                \item \textbf{Typ}: Agent-based CM, model pull
                \item \textbf{Licencja}: Apache 2.0
                \item \textbf{J\k{e}zyk}: Ruby DSL
            \end{itemize}
            \vspace{0.4em}
            \textbf{Kluczowe funkcje:}
            \begin{itemize}
                \item Chef Server, Workstation, Client (node)
                \item Test Kitchen -- automatyczne testy konfiguracji
                \item InSpec -- testy zgodno\'sci (compliance as code)
                \item Supermarket -- spo\l{}eczno\'sciowe cookbooki
            \end{itemize}
        \end{column}
        \begin{column}{0.34\textwidth}
            \begin{block}{Zastosowanie}
                \begin{itemize}
                    \item Z\l{}o\.zone logiki
                    \item DevSecOps
                    \item Compliance testing
                    \item Finanse i banki
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Terraform}
    \begin{columns}[T]
        \begin{column}{0.63\textwidth}
            \begin{itemize}
                \item \textbf{Rok}: 2014, HashiCorp
                \item \textbf{Typ}: IaC Provisioning, model push
                \item \textbf{Licencja}: MPL 2.0 (CE) / HCP Terraform
                \item \textbf{J\k{e}zyk}: HCL (HashiCorp Configuration Language)
            \end{itemize}
            \vspace{0.4em}
            \textbf{Kluczowe funkcje:}
            \begin{itemize}
                \item Tworzenie infrastruktury
                \item Ponad 3\,000 provider\'ow (AWS itd.)
                \item \texttt{plan} $\rightarrow$ \texttt{apply} -- podgl\k{a}d zmian przed wdro\.zeniem
                \item Zarz\k{a}dzanie stanem (\texttt{terraform.tfstate})
                \item Fork OpenTofu (BSL od 2023)
            \end{itemize}
        \end{column}
        \begin{column}{0.34\textwidth}
            \begin{block}{Zastosowanie}
                \begin{itemize}
                    \item Chmura (AWS/Azure/GCP)
                    \item Multi-cloud
                    \item Kubernetes
                    \item Sieci i DNS
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{Podsumowanie}
% ============================================================

\begin{frame}{Podsumowanie korzy\'sci}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Dla programisty / admina}
            \begin{itemize}
                \item Automatyzacja powtarzalnych zada\'n
                \item Wersjonowanie infrastruktury jak kodu
                \item Szybki rollback do poprzedniego stanu
                \item Reproducibility -- identyczne \'srodowiska
                \item Mniej pomy\l{}ek, mniej pracy r\k{e}cznej
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Dla organizacji}
            \begin{itemize}
                \item Ni\.zsze koszty operacyjne (OPEX)
                \item Szybsze wdro\.zenia -- wy\.zszy TTM
                \item Zgodno\'s\'c z regulacjami (compliance)
                \item Wy\.zsza dost\k{e}pno\'s\'c (SLA/SLO)
                \item Dokumentacja infrastruktury w kodzie
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.8em}
\end{frame}

\section{\'{C}wiczenia praktyczne}
% ============================================================

\begin{frame}[fragile]{Git -- podstawy dla \'{c}wicze\'n}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Podstawowe komendy}
            \begin{itemize}
                \item \texttt{git init} / \texttt{git clone <url>}
                \item \texttt{git add .} / \texttt{git commit -m}
                \item \texttt{git push} / \texttt{git pull}
                \item \texttt{git branch <name>} / \texttt{git checkout}
                \item \texttt{git merge} / \texttt{git rebase}
                \item \texttt{git log --oneline --graph}
                \item \texttt{git stash} / \texttt{git tag v1.0}
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Workflow GitFlow}
            \begin{itemize}
                \item \texttt{main} -- produkcja (stabilna)
                \item \texttt{develop} -- integracja
                \item \texttt{feature/X} -- nowe funkcje
                \item \texttt{release/X} -- przygotowanie
                \item \texttt{hotfix/X} -- pilne poprawki
            \end{itemize}
            \vspace{0.3em}
            \textbf{Dobre praktyki}
            \begin{itemize}
                \item Conventionals Commits
                \item \texttt{.gitignore} dla sekret\'ow
                \item Pull Requests + code review
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Ansible -- podstawy dla \'{c}wicze\'n}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Struktura projektu}
            \begin{itemize}
                \item \texttt{inventory/} -- lista host\'ow
                \item \texttt{playbooks/} -- g\l{}\'owne playbooki
                \item \texttt{roles/} -- wielokrotnego u\.zytku
                \item \texttt{group\_vars/} -- zmienne grup
                \item \texttt{ansible.cfg} -- konfiguracja
            \end{itemize}
            \vspace{0.3em}
            \textbf{Ad-hoc commands}\\[0.2em]
            \texttt{\scriptsize ansible all -m ping}\\
            \texttt{\scriptsize ansible web -m apt -a "name=nginx"}\\
            \texttt{\scriptsize ansible-playbook site.yml}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Przyk\l{}adowy playbook YAML}\\[0.2em]
            \scriptsize
            \texttt{---}\\
            \texttt{- hosts: webservers}\\
            \texttt{\ \ become: yes}\\
            \texttt{\ \ vars:}\\
            \texttt{\ \ \ \ pkg: nginx}\\
            \texttt{\ \ tasks:}\\
            \texttt{\ \ - name: Install pkg}\\
            \texttt{\ \ \ \ apt:}\\
            \texttt{\ \ \ \ \ \ name: "\{\{ pkg \}\}"}\\
            \texttt{\ \ \ \ \ \ state: present}\\
            \texttt{\ \ - name: Start pkg }\\
            \texttt{\ \ \ \ service:}\\
            \texttt{\ \ \ \ \ \ name: "\{\{ pkg \}\}"}\\
            \texttt{\ \ \ \ \ \ state: started}\\
            \texttt{\ \ \ \ \ \ enabled: yes}
        \end{column}
    \end{columns}
\end{frame}
